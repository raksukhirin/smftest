// ยอด THB ต่อบรรทัด (+ กรณี THBAMT ยังไม่ post)
function amtThb(l){
  const b = l.ITEMTYPE==='ADV' ? l.THBTOT : l.THBAMT;
  if(b!=null) return b*(l.DOCBAL||1);
  // THBAMT ว่าง → CALAMT × SALERATE
  const c = l.ITEMTYPE==='ADV' ? l.CALTOT : l.CALAMT;
  return c*(+l.SALERATE||0)*(l.DOCBAL||1);
}
const isNA     = l => l.DOCTYPE==='N/A';
const posted   = l => l.REFINVDETID!=null && l.POSTENTRY!==3;
// ธง "ไม่ให้" ที่ผู้ใช้ตั้งเอง — ไม่ใช่ BadFx

// pool ที่ให้เฉลี่ย: ตัดบรรทัดที่ติดธง "0" ออก
const allocPool = () => LN().filter(l =>
  l.HBLREF===MASTER.DOCNO && posted(l) && !excluded(l));

// {@cAllocBy}: ALLOCBY ชนะ · ว่างจึงเดาจาก UOM
function ratio(l,h){
  switch(basisOf(l)){
   case 'shp': return JOB.TOTBL? 1/JOB.TOTBL : 1;
   case 'cbm': return JOB.REVTON? h.REVTON/JOB.REVTON : 0;
   case 'kgs': return JOB.GW? h.GW/JOB.GW : 0;
   case 'do' : return JOB.TOTDO? h.TOTDO/JOB.TOTDO : 0;
  }
}
